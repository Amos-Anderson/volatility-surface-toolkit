\begin{frame}{The Black-Scholes Benchmark}
\framesubtitle{How wrong is a single flat volatility?}

\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{What Black-Scholes assumes:}
\begin{itemize}
    \item One constant $\sigma$ for \textbf{all} strikes and maturities
    \item Log-normal returns: no skew, no smile, no term structure
    \item Complete market: unique risk-neutral measure $\Qmeas$
\end{itemize}
\vspace{0.3cm}
\textbf{Best single volatility (least-squares):}
\begin{equation}
\hat\sigma_{\text{flat}} \;=\; \arg\min_{\sigma} \sum_{i=1}^{877} \bigl( C_{\text{BS}}^{(i)}(\sigma) - C_{\text{mid}}^{(i)} \bigr)^2
\;=\; \boxed{16.20\%}
\end{equation}
\vspace{0.2cm}
Overall pricing error: $\text{MAPE} = 8.32\%$
\end{column}

\begin{column}{0.42\textwidth}
\begin{block}{The Benchmark Logic}
We do not advocate BS. We use it as a \textbf{deliberately naive floor}. Any model that cannot beat $8.32\%$ MAPE is not economically viable.
\end{block}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Generation 1: Heston Stochastic Volatility}
\framesubtitle{Making volatility itself a random process}

\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{The Heston dynamics:}
\begin{align*}
\mathrm{d}S_t &= (r-q)S_t\,\mathrm{d}t + \sqrt{v_t}\,S_t\,\mathrm{d}W_t^1\\
\mathrm{d}v_t &= \kappa(\theta-v_t)\,\mathrm{d}t + \sigma_v\sqrt{v_t}\,\mathrm{d}W_t^2
\end{align*}
with $\mathrm{d}\langle W^1,W^2\rangle = \rho\,\mathrm{d}t$.

\vspace{0.3cm}
\textbf{The economic mechanism:}
\begin{itemize}
    \item $\rho < 0$: spot drops $\Rightarrow$ variance rises $\Rightarrow$ OTM puts expensive
    \item CIR: variance mean-reverts to $\theta$ at speed $\kappa$
    \item Closed-form characteristic function $\Rightarrow$ FFT pricing
\end{itemize}
\end{column}

\begin{column}{0.42\textwidth}
\begin{block}{Fitted Parameters (GLD)}
\begin{tabular}{@{}lc@{}}\toprule
Parameter & Value\\
\midrule
$v_0$ (initial var) & 0.0220\\
$\kappa$ (speed) & 10.00\\
$\theta$ (long var) & 0.0308\\
$\sigma_v$ (vol of vol) & 0.6598\\
$\rho$ (correlation) & +0.24\\
\bottomrule
\end{tabular}
\\[0.2cm]
\footnotesize Feller: $2\kappa\theta - \sigma_v^2 = 0.18 > 0$ \checkmark
\end{block}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Generation 1: The SVJ Extension}
\framesubtitle{Adding compound Poisson jumps to Heston (Bates 1996)}
\small

\textbf{The SVJ dynamics:}
\begin{align*}
\mathrm{d}S_t &= (r-q-\lambda\mu_J)\,S_t\,\mathrm{d}t + \sqrt{v_t}\,S_t\,\mathrm{d}W_t^1 + S_t\,(e^{J}-1)\,\mathrm{d}N_t\\[2pt]
\mathrm{d}v_t &= \kappa(\theta-v_t)\,\mathrm{d}t + \sigma_v\sqrt{v_t}\,\mathrm{d}W_t^2\\[2pt]
J &\sim \mathcal{N}(\mu_J,\sigma_J^2),\qquad \mathrm{d}N_t \sim \text{Poisson}(\lambda\,\mathrm{d}t)
\end{align*}

\textbf{Calibrated jump parameters} (Heston params frozen):
\quad $\lambda = 0.0102$, \quad $\mu_J \approx 0$, \quad $\sigma_J = 0.01$

\centering
\small
\begin{tabular}{@{}lccc@{}}\toprule
\textbf{Metric} & \textbf{BS Flat Vol} & \textbf{Heston SV} & \textbf{Heston + SVJ}\\\midrule
Overall MAPE & 8.32\% & 5.83\% & \textbf{5.83\%}\\
Deep OTM put MAPE & 8.36\% & 12.19\% & \textbf{12.19\%}\\\bottomrule
\end{tabular}

\textcolor{red}{\textbf{The jump component did not activate.}}

\vspace{0.2cm}
\footnotesize
\begin{itemize}
\item With GLD's narrow strike range ($K/S \in [0.924, 1.064]$), the optimizer finds
$\lambda \approx 0$, $\mu_J \approx 0$: \emph{no jumps needed to fit the observed prices.}
\item This confirms Gatheral (2006): finite-activity Poisson jumps cannot reproduce\\[1pt]
the continuous short-dated skew. \textbf{Infinite-activity is needed}---L\'evy processes.
\end{itemize}

\end{frame}

\begin{frame}{Generation 2: L\'evy Processes}
\framesubtitle{Infinite-activity jumps to capture the short-dated skew}

\textbf{The problem Generation 1 left unsolved:}
\begin{itemize}
    \item Heston diffusion needs \emph{time} to generate variance --- short-dated skew too shallow
    \item SVJ adds \emph{finite} Poisson jumps ($\lambda$ too low) --- still not enough
    \item Empirical skew scales as $T^H$, $H < 1/2$; diffusion gives $T^{1/2}$
\end{itemize}

\vspace{0.2cm}
\textcolor{bbgorange}{\textbf{The Lévy insight:}} Replace the Gaussian underlying process with a process that has \textbf{infinite activity}---jumps at \emph{every} time scale, not just discrete arrivals.

\vspace{0.2cm}
\textbf{L\'evy-Khinchine triplet $(\gamma, \sigma, \nu)$:}
\begin{equation}
\varphi(u) = \exp\!\Bigl\{t\Bigl[i\gamma u - \frac{\sigma^2 u^2}{2} + \int_{\mathbb{R}}(e^{iux}-1-iux\mathbf{1}_{|x|<1})\,\nu(\mathrm{d}x)\Bigr]\Bigr\}
\end{equation}

\vspace{0.2cm}
\textbf{Market incompleteness:} Jumps cannot be hedged by trading $S_t$ alone. The Esscher transform selects a unique risk-neutral measure $\mathbb{Q}_\theta$ via exponential tilting.

\end{frame}

\begin{frame}{1. Normal Inverse Gaussian (NIG)}
\framesubtitle{Barndorff-Nielsen (1997) --- Brownian motion subordinated by Inverse Gaussian time}

\textbf{The intuition.} Start with standard Brownian motion with drift $\theta$:
\[
X_t \;=\; \theta t + \sigma W_t
\]
Now replace calendar time $t$ with a random clock $T_t$ that follows \emph{Inverse Gaussian} distribution (also called the Wald distribution). The resulting process is \textbf{NIG}:
\[
L_t \;=\; \theta\, T_t + \sigma\, W_{T_t}
\]

\vspace{0.2cm}
\textbf{Why this matters for options.} The Inverse Gaussian clock has \textbf{heavier tails} than the normal clock. When time itself is random and skewed, the returns inherit skewness and kurtosis even though the underlying Brownian motion is Gaussian.

\end{frame}

\begin{frame}{1. Normal Inverse Gaussian (NIG)}

\textbf{Characteristic function} (closed form):
\[
\varphi_{\text{NIG}}(u) \;=\;
\exp\!\Bigl\{
i\mu u t + \delta t\bigl(
\sqrt{\alpha^2 - \beta^2} - \sqrt{\alpha^2 - (\beta+iu)^2}
\bigr)
\Bigr\}
\]

\textbf{Parameters:} 
\quad
\begin{tabular}{@{}ll@{}}\toprule
$\alpha > 0$ & Tail heaviness (steepness of the smile)\\
$|\beta| < \alpha$ & Skewness ($\beta<0$ for negative skew)\\
$\delta > 0$ & Scale (overall volatility level)\\
$\mu \in \mathbb{R}$ & Location (drift adjustment)\\\bottomrule
\end{tabular}

\vspace{0.2cm}
\textbf{Key property:} NIG is a \emph{normal variance-mean mixture}. The density involves the modified Bessel function $K_1$, but the characteristic function above is elementary enough for FFT pricing.

\end{frame}

\begin{frame}{2. Variance Gamma (VG)}
\framesubtitle{Madan, Carr \& Chang (1998) --- Brownian motion subordinated by Gamma time}

\textbf{The intuition.} Same recipe as NIG: start with Brownian motion, but now replace the clock with a \emph{Gamma} process $G_t$ instead of Inverse Gaussian. The Gamma clock is \textbf{pure-jump with infinite activity} --- it ticks at every instant, but most ticks are infinitesimally small.

\[
L_t \;=\; \theta\, G_t + \sigma\, W_{G_t}, \qquad G_t \sim \Gamma\!\left(\frac{t}{\nu},\,\nu\right)
\]

\vspace{0.2cm}
\textbf{Why this matters for options.} The Gamma subordinator produces \textbf{finite moments but infinite activity}. The model generates a continuous stream of small jumps that creates the steep short-dated skew --- exactly what Heston diffusion misses.

\end{frame}

\begin{frame}{2. Variance Gamma (VG)}

\textbf{Characteristic function} (elementary):
\[
\varphi_{\text{VG}}(u) \;=\;
\Bigl(
1 - i\theta\nu u + \frac{\sigma^2\nu u^2}{2}
\Bigr)^{-t/\nu}
\]

\vspace{0.2cm}
\textbf{Parameters:}
\quad
\begin{tabular}{@{}ll@{}}\toprule
$\sigma > 0$ & Volatility of the Brownian motion\\
$\nu > 0$ & Variance rate of the Gamma clock\\
& \quad (as $\nu \to 0$, VG $\to$ Black-Scholes)\\
$\theta \in \mathbb{R}$ & Drift of the subordinated Brownian motion\\
& \quad ($\theta < 0$ produces negative skew)\\\bottomrule
\end{tabular}

\vspace{0.2cm}
\textbf{Key property:} VG is the \emph{simplest} exponential L\'evy model for finance --- no Bessel functions, no hypergeometrics. The CF is a power function, making calibration fast and FFT pricing trivial.

\end{frame}

\begin{frame}{3. Bilateral Gamma Motion (BGM)}
\framesubtitle{Michaelides \& Kirkby (2024) --- independent Gamma clocks for each tail}

\textbf{The intuition.} NIG and VG use a \emph{single} time change for the entire return distribution. BGM breaks the clock into two \textbf{independent} Gamma processes:
\[
L_t \;=\; X_t^+ - X_t^-, \qquad
X_t^\pm \sim \Gamma(\alpha_\pm t,\, \beta_\pm)
\]
$X^+$ drives \textbf{positive} jumps (upside), $X^-$ drives \textbf{negative} jumps (downside). They are completely decoupled.

\vspace{0.2cm}
\textbf{Why this matters for gold (GLD).} Equity indices typically exhibit \emph{one-sided} skew --- OTM puts are expensive, OTM calls are cheap. Gold is different: both supply shocks and safe-haven demand can generate large moves. A model with \textbf{independent tail parameters} can fit two-sided tail risk without forcing symmetry.

\end{frame}

\begin{frame}{3. Bilateral Gamma Motion (BGM)}

\textbf{Characteristic function} (product of two Gamma CFs):
\[
\varphi_{\text{BGM}}(u) \;=\;
\Bigl(1 - \frac{iu}{\beta_+}\Bigr)^{-\alpha_+ t}
\Bigl(1 + \frac{iu}{\beta_-}\Bigr)^{-\alpha_- t}
\]

\vspace{0.2cm}
\textbf{Parameters:}
\quad
\begin{tabular}{@{}ll@{}}\toprule
$\alpha_+, \alpha_-$ & Shape (tail heaviness of upside / downside)\\
$\beta_+, \beta_-$ & Rate (frequency of positive / negative jumps)\\
& \quad Larger $\beta_+$ $\Rightarrow$ fewer large upward jumps\\
& \quad Smaller $\beta_-$ $\Rightarrow$ fatter left tail (crash risk)\\\bottomrule
\end{tabular}

\vspace{0.2cm}
\textbf{Key property:} BGM is the only model among the three where \textbf{skewness and kurtosis are separately controlled for each tail}. This is economically meaningful for commodities and cryptocurrencies, where two-sided tail risk is the rule, not the exception.

\end{frame}

\begin{frame}{Generation 2: Expected vs. Actual Performance}
  \textbf{Literature expectation} (Carr--Madan 2002, Schoutens 2003).
  On equity-index options (SPX), infinite-activity Lévy processes typically
  \emph{beat} stochastic-volatility models on the short-dated skew because
  semi-heavy tails generate a steeper OTM-put smile than a CIR variance
  process alone.

  \vspace{0.2cm}
  \textbf{What we found on GLD (2025-07-18).}

  \begin{center}
  \begin{tabular}{lccc}
    \toprule
    \textbf{Model} & \textbf{IV-MAPE (\%)} & \textbf{Price-MAPE (\%)} & \textbf{RMSE (\$)} \\
    \midrule
    Black--Scholes (flat vol)    & 8.32  & 8.32  & —     \\
    Heston (stochastic vol)      & \textbf{5.83}  & \textbf{5.83}  & —     \\
    NIG  (infinite-activity)     & 8.22  & 17.66 & 0.29  \\
    VG   (infinite-activity)     & 8.37  & 13.58 & 0.36  \\
    BGM  (infinite-activity)     & 8.35  & 18.96 & 0.29  \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.1cm}
  \textbf{No improvement.}  All three Lévy models match (but do not beat)
  flat-vol BS in IV space, and are significantly \emph{worse} than Heston.
\end{frame}

% ---------------------------------------------------------------------------
% Frame 2: Why the Lévy models found no marginal gain
% ---------------------------------------------------------------------------
\begin{frame}{Why Lévy underperforms on GLD: three explanations}
  \begin{enumerate}
    \item \textbf{Boundary-seeking = Gaussian collapse.}
    Differential evolution pushes parameters to box edges:
    \[
      \text{NIG: }\alpha=24.7\;(\text{ub}=25),\quad
      \text{VG: }\nu=0.01\;(\text{lb}),\quad
      \text{BGM: }\alpha_-=20\;(\text{ub}).
    \]
    As $\alpha\to\infty$ (NIG) or $\nu\to 0$ (VG), the process collapses to
    Brownian motion.  The data is telling the optimizer: ``I do not need jumps.''

    \vspace{0.3cm}
    \item \textbf{Narrow moneyness range.}
    GLD strikes span only $K/S \in [0.924, 1.064]$.  There are no deep OTM
    wings ($K/S < 0.8$ or $> 1.2$) where semi-heavy tails would generate the
    characteristic steep skew that Lévy models are famous for.

    \vspace{0.3cm}
    \item \textbf{IV-space vs. Price-space pathology.}
    In IV space the Lévy models match BS ($\sim$8\%).  In price space they
    degrade to 13--19\% because deep OTM puts have prices $\ll \$1$; tiny
    dollar errors become enormous relative errors when the denominator is small.
  \end{enumerate}
\end{frame}

\begin{frame}{The empirical bridge to Generation 3}
  \textbf{What we have established so far.}
  \begin{itemize}
    \item BS flat-vol fails (8.32\% MAPE) --- the smile exists.
    \item Heston captures it (5.83\% MAPE) --- stochastic volatility is
      the \emph{dominant} mechanism for this surface.
    \item Lévy adds no marginal value (8.2--8.4\% IV-MAPE) --- the data
      does not exhibit the extreme kurtosis or wide moneyness wings that
      infinite-activity jumps were designed for.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{The unanswered question.}
  If parametric models (BS, Heston, NIG, VG, BGM) are all \emph{too rigid}
  or \emph{too expressive} for this surface, can we learn the mapping
  $(K, T) \mapsto \sigma_{\text{impl}}$ directly from the data without
  imposing a functional form?

  \vspace{0.3cm}
  \textbf{Generation 3: Deep Learning.}
  \begin{itemize}
    \item \textbf{ConvLSTM} --- spatiotemporal evolution of the surface.
    \item \textbf{Attention-LSTM} --- cross-dependencies across strikes
      and maturities.
    \item \textbf{Physics-informed Transformer} --- hard-wires no-arbitrage
      constraints (calendar spread, butterfly) into the architecture.
  \end{itemize}
\end{frame}
